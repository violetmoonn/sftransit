# Checks for new transit alerts every 10 minutes and emails subscribers.
# (Vercel's free plan only allows once-a-day cron jobs; GitHub Actions is free for public repos.)
# Setup: repo Settings → Secrets and variables → Actions → New repository secret
#   CRON_SECRET = the same value you set in Vercel
name: Send transit alerts
on:
  schedule:
    - cron: "*/10 * * * *"
  workflow_dispatch: {}
jobs:
  send:
    runs-on: ubuntu-latest
    timeout-minutes: 3
    steps:
      - name: Trigger alert check
        run: |
          curl -sS --fail-with-body --max-time 60 \
            -H "Authorization: Bearer ${{ secrets.CRON_SECRET }}" \
            https://sf-transit.com/api/send-alerts
